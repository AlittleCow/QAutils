代码:STR2CON(CODE),NODRAW;
周期:PERIOD+1,NODRAW;

KMODE:1,NODRAW;  {K线发送模式: 1=发送单个K线数据(TDXDLL1 16), 0=发送K线序列数据(TDXDLL1 20)}

DUMMY1:=TDXDLL1(1,O,H,L);  {TDXKBAR_SETOHLC - 设置开高低价格数据}

{STEP 2: 设置日期时间数据}
DUMMY3:=TDXDLL1(3,YEAR,MONTH,DAY);     {TDXKBAR_SETDATE - 设置日期}
DUMMY4:=TDXDLL1(4,HOUR,MINUTE,代码周期);   {TDXKBAR_SETTIMEANDFINALIZE - 设置时间并完成数据准备，使用代码周期作为编码}

{STEP 3: 测试ZEROMQ服务器API功能 - 基础服务器操作}
客户端时间:TIME,NODRAW;
心跳响应:=TDXDLL1(10,客户端时间,0,0);    {TDXSERVER_HEARTBEAT - ZEROMQ服务器心跳测试}

{根据KMODE控制K线数据发送方式}
K线发送结果:=IF(KMODE=1,TDXDLL1(16,代码周期,BARSLEN,0),0);  {TDXSERVER_SENDKBAR - 发送单个K线数据到QUANTSERVER}
K线序列发送:=IF(KMODE=0,TDXDLL1(20,代码周期,0, BARSLEN),0);  {TDXSERVER_SENDKBARSERIES - 发送K线序列到QUANTSERVER}
